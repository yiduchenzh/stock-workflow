"""
融合 K 线形态推演 + 关键价位 + 缠论的交易推理系统
"""

from dataclasses import dataclass, field
from typing import List, Optional, Tuple, Dict
from enum import Enum
import math


# ========== 数据结构 ==========

@dataclass
class Bar:
    """单根 K 线"""
    dt: str  # 时间戳
    o: float
    h: float
    l: float
    c: float
    v: float = 0.0

    @property
    def upper_shadow(self) -> float:
        return self.h - max(self.o, self.c)

    @property
    def lower_shadow(self) -> float:
        return min(self.o, self.c) - self.l

    @property
    def body(self) -> float:
        return abs(self.c - self.o)

    @property
    def amplitude(self) -> float:
        return self.h - self.l

    @property
    def direction(self) -> str:
        if self.c > self.o:
            return "阳"
        elif self.c < self.o:
            return "阴"
        return "十字"


# ========== K 线 12 形态分类 ==========

class KLineShape(Enum):
    BULL_MARUBOZU = "光头光脚阳线"
    BULL_LOWER = "带下影阳线"
    BULL_UPPER = "带上影阳线"
    BULL_BOTH = "带上下影阳线"
    BEAR_MARUBOZU = "光头光脚阴线"
    BEAR_LOWER = "带下影阴线"
    BEAR_UPPER = "带上影阴线"
    BEAR_BOTH = "带上下影阴线"
    DOJI_FLAT = "一字线"
    DOJI_T = "T字线"
    DOJI_INV_T = "倒T字线"
    DOJI = "十字星"


def classify_bar(bar: Bar, eps: float = 1e-9) -> KLineShape:
    """将单根 K 线分类到 12 种基本形态之一"""
    u = bar.upper_shadow
    d = bar.lower_shadow
    has_u = u > eps
    has_d = d > eps
    direction = bar.direction

    if direction == "阳":
        if not has_u and not has_d:
            return KLineShape.BULL_MARUBOZU
        if not has_u and has_d:
            return KLineShape.BULL_LOWER
        if has_u and not has_d:
            return KLineShape.BULL_UPPER
        return KLineShape.BULL_BOTH
    elif direction == "阴":
        if not has_u and not has_d:
            return KLineShape.BEAR_MARUBOZU
        if not has_u and has_d:
            return KLineShape.BEAR_LOWER
        if has_u and not has_d:
            return KLineShape.BEAR_UPPER
        return KLineShape.BEAR_BOTH
    else:  # 十字
        if not has_u and not has_d:
            return KLineShape.DOJI_FLAT
        if not has_u and has_d:
            return KLineShape.DOJI_T
        if has_u and not has_d:
            return KLineShape.DOJI_INV_T
        return KLineShape.DOJI

# ========== 关键价位计算 ==========

@dataclass
class KeyLevels:
    """基于前日 OHLC 推算的今日关键价位"""
    prev_o: float
    prev_h: float
    prev_l: float
    prev_c: float

    # 基本位置
    entity_top: float = 0.0
    entity_bottom: float = 0.0
    entity_mid: float = 0.0
    range_mid: float = 0.0
    range: float = 0.0

    # 斐波那契
    fib_382: float = 0.0
    fib_500: float = 0.0
    fib_618: float = 0.0
    fib_786: float = 0.0
    fib_ext_1272: float = 0.0
    fib_ext_1618: float = 0.0

    # 枢轴
    pivot: float = 0.0
    r1: float = 0.0
    r2: float = 0.0
    r3: float = 0.0
    s1: float = 0.0
    s2: float = 0.0
    s3: float = 0.0

    # 涨跌停（默认主板 10%）
    limit_up: float = 0.0
    limit_down: float = 0.0

    def compute(self, limit_pct: float = 0.10):
        po, ph, pl, pc = self.prev_o, self.prev_h, self.prev_l, self.prev_c
        self.entity_top = max(po, pc)
        self.entity_bottom = min(po, pc)
        self.entity_mid = (po + pc) / 2
        self.range_mid = (ph + pl) / 2
        self.range = ph - pl

        # 斐波那契回撤
        self.fib_382 = pl + 0.382 * self.range
        self.fib_500 = pl + 0.500 * self.range
        self.fib_618 = pl + 0.618 * self.range
        self.fib_786 = pl + 0.786 * self.range
        self.fib_ext_1272 = pl + 1.272 * self.range
        self.fib_ext_1618 = pl + 1.618 * self.range

        # 经典枢轴
        p = (ph + pl + pc) / 3
        self.pivot = p
        self.r1 = 2 * p - pl
        self.s1 = 2 * p - ph
        self.r2 = p + (ph - pl)
        self.s2 = p - (ph - pl)
        self.r3 = ph + 2 * (p - pl)
        self.s3 = pl - 2 * (ph - p)

        # 涨跌停
        self.limit_up = round(pc * (1 + limit_pct), 2)
        self.limit_down = round(pc * (1 - limit_pct), 2)
        return self

    def as_dict(self):
        return {
            "前高 PH": self.prev_h, "前收 PC": self.prev_c,
            "前开 PO": self.prev_o, "前低 PL": self.prev_l,
            "实体上沿": self.entity_top, "实体下沿": self.entity_bottom,
            "实体中点": self.entity_mid, "区间中点": self.range_mid,
            "Fib 0.382": self.fib_382, "Fib 0.500": self.fib_500,
            "Fib 0.618": self.fib_618, "Fib 0.786": self.fib_786,
            "Fib Ext 1.272": self.fib_ext_1272, "Fib Ext 1.618": self.fib_ext_1618,
            "Pivot": self.pivot, "R1": self.r1, "R2": self.r2, "R3": self.r3,
            "S1": self.s1, "S2": self.s2, "S3": self.s3,
            "涨停": self.limit_up, "跌停": self.limit_down,
        }

# ========== 多周期聚合 ==========

def aggregate(bars: List[Bar], n: int) -> List[Bar]:
    """将 n 根小周期 K 线合成一根大周期 K 线"""
    result = []
    for i in range(0, len(bars) - n + 1, n):
        chunk = bars[i:i+n]
        result.append(Bar(
            dt=chunk[0].dt,
            o=chunk[0].o,
            h=max(b.h for b in chunk),
            l=min(b.l for b in chunk),
            c=chunk[-1].c,
            v=sum(b.v for b in chunk),
        ))
    return result


# ========== 缠论：包含处理 ==========

@dataclass
class MergedBar:
    """经过包含处理后的合并 K 线"""
    dt_start: str
    dt_end: str
    h: float
    l: float
    direction: int  # 1 向上合并，-1 向下合并
    raw: Bar


def merge_inclusion(bars: List[Bar]) -> List[MergedBar]:
    """缠论包含处理"""
    if not bars:
        return []
    
    merged = [MergedBar(bars[0].dt, bars[0].dt, bars[0].h, bars[0].l, 1, bars[0])]
    direction = 1

    for bar in bars[1:]:
        last = merged[-1]
        # 判断是否包含
        contains = (bar.h <= last.h and bar.l >= last.l) or \
                   (bar.h >= last.h and bar.l <= last.l)
        if not contains:
            # 无包含，确定方向
            if bar.h > last.h and bar.l > last.l:
                direction = 1
            elif bar.h < last.h and bar.l < last.l:
                direction = -1
            merged.append(MergedBar(bar.dt, bar.dt, bar.h, bar.l, direction, bar))
        else:
            # 有包含，按方向合并
            if direction == 1:
                new_h = max(last.h, bar.h)
                new_l = max(last.l, bar.l)
            else:
                new_h = min(last.h, bar.h)
                new_l = min(last.l, bar.l)
            merged[-1] = MergedBar(last.dt_start, bar.dt, new_h, new_l, direction, bar)
    return merged

# ========== 缠论：分型 ==========

class FractalType(Enum):
    TOP = "顶分型"
    BOTTOM = "底分型"


@dataclass
class Fractal:
    type: FractalType
    idx: int  # 在合并 K 线序列中的索引
    h: float
    l: float
    dt: str


def find_fractals(merged: List[MergedBar]) -> List[Fractal]:
    """识别顶底分型（使用合并后 K 线）"""
    fractals = []
    for i in range(1, len(merged) - 1):
        prev, cur, nxt = merged[i-1], merged[i], merged[i+1]
        # 顶分型：中间 K 线高点最高，低点也最高
        if cur.h > prev.h and cur.h > nxt.h and cur.l > prev.l and cur.l > nxt.l:
            fractals.append(Fractal(FractalType.TOP, i, cur.h, cur.l, cur.dt_end))
        # 底分型：中间 K 线低点最低，高点也最低
        elif cur.l < prev.l and cur.l < nxt.l and cur.h < prev.h and cur.h < nxt.h:
            fractals.append(Fractal(FractalType.BOTTOM, i, cur.h, cur.l, cur.dt_end))
    return fractals


# ========== 缠论：笔 ==========

@dataclass
class Stroke:
    """笔：相邻的顶分型和底分型之间的连接"""
    start: Fractal
    end: Fractal
    direction: int  # 1 向上，-1 向下


def build_strokes(fractals: List[Fractal], min_bars: int = 4) -> List[Stroke]:
    """构造笔（简化版：相邻顶底交替，且间隔至少 min_bars 根合并 K 线）"""
    if len(fractals) < 2:
        return []
    
    strokes = []
    # 过滤出有效分型序列：顶底交替
    valid = [fractals[0]]
    for f in fractals[1:]:
        if f.type != valid[-1].type:
            # 同类型相邻，取更极端的
            if len(valid) >= 2 and f.type == valid[-2].type:
                # 比较保留更极端
                if f.type == FractalType.TOP and f.h > valid[-2].h:
                    valid[-1] = f
                elif f.type == FractalType.BOTTOM and f.l < valid[-2].l:
                    valid[-1] = f
            else:
                valid.append(f)
        else:
            # 同类型，保留更极端的
            if f.type == FractalType.TOP and f.h > valid[-1].h:
                valid[-1] = f
            elif f.type == FractalType.BOTTOM and f.l < valid[-1].l:
                valid[-1] = f
    
    for i in range(len(valid) - 1):
        a, b = valid[i], valid[i+1]
        if abs(b.idx - a.idx) >= min_bars:
            direction = 1 if b.type == FractalType.TOP else -1
            strokes.append(Stroke(a, b, direction))
    return strokes

# ========== 缠论：中枢 ==========

@dataclass
class Center:
    """中枢：至少三个连续次级别走势重叠部分"""
    zg: float  # 中枢上沿
    zd: float  # 中枢下沿
    start_idx: int
    end_idx: int
    strokes: List[Stroke] = field(default_factory=list)

    @property
    def height(self) -> float:
        return self.zg - self.zd


def find_centers(strokes: List[Stroke], min_strokes: int = 3) -> List[Center]:
    """中枢识别：连续 3 笔以上的重叠区间"""
    centers = []
    i = 0
    while i + min_strokes - 1 < len(strokes):
        group = strokes[i:i+min_strokes]
        # 计算重叠区间
        highs = []
        lows = []
        for s in group:
            highs.append(max(s.start.h, s.end.h))
            lows.append(min(s.start.l, s.end.l))
        zg = min(highs)  # 中枢上沿 = 各笔高点中的最小值
        zd = max(lows)   # 中枢下沿 = 各笔低点中的最大值
        if zg > zd:
            centers.append(Center(zg, zd, group[0].start.idx, group[-1].end.idx, group))
            i += 1
        else:
            i += 1
    return centers

# ========== 形态推演引擎 ==========

class ShapeInference:
    """根据当前价格推演大周期最终形态候选集"""

    def __init__(self, target_bars: int, o: float, h_t: float, l_t: float, c_t: float, bars_done: int):
        self.o = o
        self.h_t = h_t
        self.l_t = l_t
        self.c_t = c_t
        self.bars_done = bars_done
        self.target_bars = target_bars

    def infer_candidates(self) -> List[KLineShape]:
        """
        根据当前 o/h/l/c 以及剩余时间，给出候选形态
        简化逻辑：
        - 如果极值已锁定（时间已尽），唯一确定
        - 否则根据剩余可能的波动方向枚举
        """
        from itertools import product

        candidates = set()
        remaining = self.target_bars - self.bars_done

        # 如果已经走完，唯一确定
        if remaining <= 0:
            return [classify_bar(Bar("", self.o, self.h_t, self.l_t, self.c_t))]

        # 枚举剩余可能：未来最高可能 >= h_t，未来最低可能 <= l_t
        # 收盘可能在任何位置
        # 这里用连续假设，判断最终可能落入哪些形态类别
        # 简化：考虑 future_h 可选 h_t 或更高，future_l 可选 l_t 或更低
        possible_h = [self.h_t]
        possible_l = [self.l_t]

        # 如果还有时间，理论上可能创新高/新低（保守枚举）
        # 这里的 epsilon 可以按波动率替换
        eps = (self.h_t - self.l_t) * 0.1 + 1e-9
        possible_h.append(self.h_t + eps)
        possible_l.append(self.l_t - eps)

        for fh, fl in product(possible_h, possible_l):
            if fh < fl:
                continue
            # 收盘价可能在整个区间内
            # 为了分类，尝试几个代表性收盘价
            for fc in [self.c_t, fh, fl, (fh+fl)/2, self.o, (self.o+fh)/2, (self.o+fl)/2]:
                if not (fl <= fc <= fh):
                    continue
                if fc <= 0:
                    continue
                bar = Bar("", self.o, fh, fl, fc)
                candidates.add(classify_bar(bar))

        # 加入当前暂定形态
        candidates.add(classify_bar(Bar("", self.o, self.h_t, self.l_t, self.c_t)))
        return sorted(candidates, key=lambda x: x.value)

更好的做法是：给定当前的 O、H、L、C，以及我们是否认为还会出现新的极值，直接确定哪些形态是可能的。

其实，对于给定 O 和当前的 H_t、L_t、C_t，如果剩余时间 H 可以 ≥ H_t、L 可以 ≤ L_t、C 可以落在任何位置：

最终 H 可以 ≥ H_t，最终 L 可以 ≤ L_t

最终 O 固定

最终 C 可以落在 [L, H] 内

所以我们需要判断：给定 O 固定，是否存在最终 (H, L, C)，满足 L ≤ H，O ∈ [L,H]，C ∈ [L,H]，H ≥ H_t，L ≤ L_t，使得它分类为形态 X。

对于 12 种形态中的每一种，我们都可以检查是否存在这样的 (H, L, C)。

def infer_candidates(o, h_t, l_t, c_t, h_max=None, l_min=None, samples=21):
    """
    根据当前状态推演最终可能形态。
    h_max: 剩余时间可能达到的最高（None 表示无限）
    l_min: 剩余时间可能达到的最低（None 表示无限）
    """
    # 如果没给边界，用当前极值稍作延展
    if h_max is None:
        h_max = h_t + (h_t - l_t)  # 假设最多再波动一个当前振幅
    if l_min is None:
        l_min = l_t - (h_t - l_t)
    
    candidates = set()
    # 最终 H ∈ [h_t, h_max], L ∈ [l_min, l_t], C ∈ [L, H]
    # 采样
    hs = [h_t + (h_max - h_t) * i / (samples - 1) for i in range(samples)]
    ls = [l_min + (l_t - l_min) * i / (samples - 1) for i in range(samples)]
    
    for fh in hs:
        for fl in ls:
            if fh < fl:
                continue
            if not (fl <= o <= fh):
                continue
            for fc in [fl, (fl+fh)/2, fh, c_t, o]:
                if fl <= fc <= fh:
                    try:
                        candidates.add(classify_bar(Bar("", o, fh, fl, fc)))
                    except:
                        pass
    return sorted(candidates, key=lambda x: x.value)

# -*- coding: utf-8 -*-
"""
融合系统：
  前日OHLC → 关键位 → 今日形态候选集
  + 多周期聚合 + 缠论结构（包含、分型、笔、中枢）
"""

from dataclasses import dataclass, field
from typing import List, Optional, Dict, Tuple
from enum import Enum
from itertools import product
import math


# =========================================================
# 1. 基础数据结构
# =========================================================

@dataclass
class Bar:
    """单根 K 线"""
    dt: str
    o: float
    h: float
    l: float
    c: float
    v: float = 0.0

    @property
    def upper_shadow(self) -> float:
        return self.h - max(self.o, self.c)

    @property
    def lower_shadow(self) -> float:
        return min(self.o, self.c) - self.l

    @property
    def body(self) -> float:
        return abs(self.c - self.o)

    @property
    def amplitude(self) -> float:
        return self.h - self.l

    @property
    def direction(self) -> str:
        if self.c > self.o:
            return "阳"
        if self.c < self.o:
            return "阴"
        return "十字"


# =========================================================
# 2. K线 12 基本形态
# =========================================================

class KLineShape(Enum):
    BULL_MARUBOZU = "光头光脚阳线"
    BULL_LOWER    = "带下影阳线"
    BULL_UPPER    = "带上影阳线"
    BULL_BOTH     = "带上下影阳线"
    BEAR_MARUBOZU = "光头光脚阴线"
    BEAR_LOWER    = "带下影阴线"
    BEAR_UPPER    = "带上影阴线"
    BEAR_BOTH     = "带上下影阴线"
    DOJI_FLAT     = "一字线"
    DOJI_T        = "T字线"
    DOJI_INV_T    = "倒T字线"
    DOJI          = "十字星"


def classify_bar(bar: Bar, eps: float = 1e-9) -> KLineShape:
    """单根 K 线 → 12 形态之一"""
    u = bar.upper_shadow
    d = bar.lower_shadow
    has_u = u > eps
    has_d = d > eps

    if bar.c > bar.o:
        if not has_u and not has_d: return KLineShape.BULL_MARUBOZU
        if not has_u and has_d:     return KLineShape.BULL_LOWER
        if has_u and not has_d:     return KLineShape.BULL_UPPER
        return KLineShape.BULL_BOTH
    if bar.c < bar.o:
        if not has_u and not has_d: return KLineShape.BEAR_MARUBOZU
        if not has_u and has_d:     return KLineShape.BEAR_LOWER
        if has_u and not has_d:     return KLineShape.BEAR_UPPER
        return KLineShape.BEAR_BOTH
    # 十字
    if not has_u and not has_d: return KLineShape.DOJI_FLAT
    if not has_u and has_d:     return KLineShape.DOJI_T
    if has_u and not has_d:     return KLineShape.DOJI_INV_T
    return KLineShape.DOJI


# =========================================================
# 3. 前日 OHLC → 今日关键位
# =========================================================

@dataclass
class KeyLevels:
    prev_o: float
    prev_h: float
    prev_l: float
    prev_c: float

    # 基本
    entity_top: float = 0.0
    entity_bottom: float = 0.0
    entity_mid: float = 0.0
    range_mid: float = 0.0
    rng: float = 0.0

    # 斐波那契
    fib_382: float = 0.0
    fib_500: float = 0.0
    fib_618: float = 0.0
    fib_786: float = 0.0
    fib_ext_1272: float = 0.0
    fib_ext_1618: float = 0.0

    # 枢轴
    pivot: float = 0.0
    r1: float = 0.0; r2: float = 0.0; r3: float = 0.0
    s1: float = 0.0; s2: float = 0.0; s3: float = 0.0

    # 涨跌停
    limit_up: float = 0.0
    limit_down: float = 0.0

    def compute(self, limit_pct: float = 0.10) -> "KeyLevels":
        po, ph, pl, pc = self.prev_o, self.prev_h, self.prev_l, self.prev_c
        self.entity_top = max(po, pc)
        self.entity_bottom = min(po, pc)
        self.entity_mid = (po + pc) / 2
        self.range_mid = (ph + pl) / 2
        self.rng = ph - pl
        r = self.rng

        self.fib_382 = pl + 0.382 * r
        self.fib_500 = pl + 0.500 * r
        self.fib_618 = pl + 0.618 * r
        self.fib_786 = pl + 0.786 * r
        self.fib_ext_1272 = pl + 1.272 * r
        self.fib_ext_1618 = pl + 1.618 * r

        p = (ph + pl + pc) / 3
        self.pivot = p
        self.r1 = 2 * p - pl
        self.s1 = 2 * p - ph
        self.r2 = p + (ph - pl)
        self.s2 = p - (ph - pl)
        self.r3 = ph + 2 * (p - pl)
        self.s3 = pl - 2 * (ph - p)

        self.limit_up = round(pc * (1 + limit_pct), 2)
        self.limit_down = round(pc * (1 - limit_pct), 2)
        return self

    def as_dict(self) -> Dict[str, float]:
        return {
            "前高 PH": self.prev_h, "前收 PC": self.prev_c,
            "前开 PO": self.prev_o, "前低 PL": self.prev_l,
            "实体上沿": self.entity_top, "实体下沿": self.entity_bottom,
            "实体中点": self.entity_mid, "区间中点": self.range_mid,
            "Fib 0.382": self.fib_382, "Fib 0.500": self.fib_500,
            "Fib 0.618": self.fib_618, "Fib 0.786": self.fib_786,
            "Fib Ext 1.272": self.fib_ext_1272, "Fib Ext 1.618": self.fib_ext_1618,
            "Pivot": self.pivot,
            "R1": self.r1, "R2": self.r2, "R3": self.r3,
            "S1": self.s1, "S2": self.s2, "S3": self.s3,
            "涨停": self.limit_up, "跌停": self.limit_down,
        }


# =========================================================
# 4. 多周期聚合
# =========================================================

def aggregate(bars: List[Bar], n: int) -> List[Bar]:
    """n 根小周期 K → 1 根大周期 K"""
    out = []
    for i in range(0, len(bars) - n + 1, n):
        chunk = bars[i:i+n]
        out.append(Bar(
            dt=chunk[0].dt,
            o=chunk[0].o,
            h=max(b.h for b in chunk),
            l=min(b.l for b in chunk),
            c=chunk[-1].c,
            v=sum(b.v for b in chunk),
        ))
    return out

# 5. 缠论：包含处理 / 分型 / 笔 / 中枢
# =========================================================

@dataclass
class MergedBar:
    dt_start: str
    dt_end: str
    h: float
    l: float
    direction: int   # 1 向上，-1 向下
    raw: Bar


def merge_inclusion(bars: List[Bar]) -> List[MergedBar]:
    """缠论包含处理"""
    if not bars:
        return []
    merged = [MergedBar(bars[0].dt, bars[0].dt, bars[0].h, bars[0].l, 1, bars[0])]
    direction = 1

    for bar in bars[1:]:
        last = merged[-1]
        contained = (bar.h <= last.h and bar.l >= last.l) or \
                    (bar.h >= last.h and bar.l <= last.l)
        if not contained:
            if bar.h > last.h and bar.l > last.l:
                direction = 1
            elif bar.h < last.h and bar.l < last.l:
                direction = -1
            merged.append(MergedBar(bar.dt, bar.dt, bar.h, bar.l, direction, bar))
        else:
            if direction == 1:
                nh = max(last.h, bar.h)
                nl = max(last.l, bar.l)
            else:
                nh = min(last.h, bar.h)
                nl = min(last.l, bar.l)
            merged[-1] = MergedBar(last.dt_start, bar.dt, nh, nl, direction, bar)
    return merged


class FractalType(Enum):
    TOP = "顶分型"
    BOTTOM = "底分型"


@dataclass
class Fractal:
    type: FractalType
    idx: int
    h: float
    l: float
    dt: str


def find_fractals(merged: List[MergedBar]) -> List[Fractal]:
    out = []
    for i in range(1, len(merged) - 1):
        p, c, n = merged[i-1], merged[i], merged[i+1]
        if c.h > p.h and c.h > n.h and c.l > p.l and c.l > n.l:
            out.append(Fractal(FractalType.TOP, i, c.h, c.l, c.dt_end))
        elif c.l < p.l and c.l < n.l and c.h < p.h and c.h < n.h:
            out.append(Fractal(FractalType.BOTTOM, i, c.h, c.l, c.dt_end))
    return out


@dataclass
class Stroke:
    start: Fractal
    end: Fractal
    direction: int  # 1 向上，-1 向下


def build_strokes(fractals: List[Fractal], min_gap: int = 3) -> List[Stroke]:
    """构造笔：顶底交替，间隔至少 min_gap"""
    if not fractals:
        return []

    # 同类型合并，保留更极端
    filt = [fractals[0]]
    for f in fractals[1:]:
        last = filt[-1]
        if f.type == last.type:
            if f.type == FractalType.TOP and f.h > last.h:
                filt[-1] = f
            elif f.type == FractalType.BOTTOM and f.l < last.l:
                filt[-1] = f
        else:
            filt.append(f)

    strokes = []
    i = 0
    while i < len(filt) - 1:
        a, b = filt[i], filt[i+1]
        if abs(b.idx - a.idx) >= min_gap:
            strokes.append(Stroke(a, b, 1 if b.type == FractalType.TOP else -1))
            i += 1
        else:
            # 间隔太短，跳过当前分型
            i += 1
    return strokes


@dataclass
class Center:
    zg: float   # 上沿
    zd: float   # 下沿
    start_idx: int
    end_idx: int
    strokes: List[Stroke] = field(default_factory=list)

    @property
    def height(self) -> float:
        return self.zg - self.zd

    def contains(self, price: float) -> bool:
        return self.zd <= price <= self.zg


def find_centers(strokes: List[Stroke], min_strokes: int = 3) -> List[Center]:
    """中枢：连续 min_strokes 笔以上的重叠区间"""
    centers = []
    i = 0
    while i + min_strokes - 1 < len(strokes):
        group = strokes[i:i+min_strokes]
        highs = [max(s.start.h, s.end.h) for s in group]
        lows  = [min(s.start.l, s.end.l) for s in group]
        zg = min(highs)
        zd = max(lows)
        if zg > zd:
            centers.append(Center(zg, zd, group[0].start.idx, group[-1].end.idx, group))
        i += 1
    return centers

# 6. 今日形态候选集推演
# =========================================================

def infer_candidates(
    o: float,
    h_t: float,
    l_t: float,
    c_t: float,
    h_max: Optional[float] = None,
    l_min: Optional[float] = None,
    samples: int = 25,
) -> List[KLineShape]:
    """
    根据当前状态推演最终可能形态集合。
      o    : 今日开盘价（已固定）
      h_t  : 当前最高
      l_t  : 当前最低
      c_t  : 当前价
      h_max: 剩余时间可能达到的最高（None 则默认在 h_t 上再加一个振幅）
      l_min: 剩余时间可能达到的最低（None 则默认在 l_t 下再减一个振幅）
    """
    amp = max(h_t - l_t, 1e-6)
    if h_max is None:
        h_max = h_t + amp
    if l_min is None:
        l_min = l_t - amp

    cands = set()

    def sample(a: float, b: float, n: int) -> List[float]:
        if n <= 1:
            return [a]
        return [a + (b - a) * i / (n - 1) for i in range(n)]

    hs = sample(h_t, h_max, samples)
    ls = sample(l_min, l_t, samples)

    for fh in hs:
        for fl in ls:
            if fh < fl:
                continue
            if not (fl <= o <= fh):
                continue
            # 收盘价采样：极值、中点、当前价、开盘
            cs = {fl, fh, (fl+fh)/2, c_t, o}
            for fc in cs:
                if fl <= fc <= fh:
                    cands.add(classify_bar(Bar("", o, fh, fl, fc)))

    # 一定包含当前暂定形态
    cands.add(classify_bar(Bar("", o, h_t, l_t, c_t)))
    return sorted(cands, key=lambda x: x.value)


# =========================================================
# 7. 组合：用关键位约束候选集
# =========================================================

def constrain_by_levels(
    candidates: List[KLineShape],
    o: float, h_t: float, l_t: float, c_t: float,
    levels: KeyLevels,
) -> List[KLineShape]:
    """
    用前日关键位进一步排除不可能形态。
    规则示例：
      - 若当前最低 l_t 已跌破前低 PL，则"光头光脚阳线（最低=开盘）"若开盘>PL 不可能
      - 若当前最高 h_t 已突破前高 PH，则"光头光脚阴线（最高=开盘）"若开盘<PH 不可能
      - 更多规则可扩展
    """
    out = []
    for s in candidates:
        ok = True
        # 光头光脚阳线：L == O，若 O > PL 且 l_t <= PL，则最低已低于开盘，不可能
        if s == KLineShape.BULL_MARUBOZU:
            if o > levels.prev_l and l_t <= levels.prev_l:
                ok = False
        # 光头光脚阴线：H == O，若 O < PH 且 h_t >= PH，则最高已高于开盘，不可能
        if s == KLineShape.BEAR_MARUBOZU:
            if o < levels.prev_h and h_t >= levels.prev_h:
                ok = False
        # 一字线：H == L，只有全程未波动才可能
        if s == KLineShape.DOJI_FLAT:
            if h_t > l_t:
                ok = False
        if ok:
            out.append(s)
    return out

代码逻辑说明
第一层：前日 OHLC → 今日关键位
KeyLevels.compute() 生成 22 个关键位，含实体位置、斐波那契回撤/扩展、经典枢轴 R1–R3/S1–S3、涨跌停边界。

第二层：实时价格 → 12 形态分类
classify_bar() 严格按你推演的 12 种基本形态归类：阴阳 × 上下影有无。

第三层：多周期聚合
aggregate(bars, n) 严格按缠论合成规则：首根开盘、全体极值、末根收盘。

# -*- coding: utf-8 -*-
"""
麦捷科技 300319 回测框架
数据格式：CSV，列名 [date, open, high, low, close, volume]
"""
import pandas as pd
import numpy as np
from dataclasses import dataclass
from typing import List, Optional, Dict
from enum import Enum


# =========================================================
# 1. 数据加载
# =========================================================

def load_data(csv_path: str) -> pd.DataFrame:
    """
    加载日线数据。
    CSV 应包含列：date, open, high, low, close, volume
    date 格式：YYYY-MM-DD
    """
    df = pd.read_csv(csv_path, parse_dates=["date"])
    df = df.sort_values("date").reset_index(drop=True)
    df.columns = [c.lower().strip() for c in df.columns]
    return df


# =========================================================
# 2. 核心数据结构
# =========================================================

@dataclass
class Bar:
    dt: str
    o: float
    h: float
    l: float
    c: float
    v: float = 0.0

    @property
    def upper_shadow(self): return self.h - max(self.o, self.c)
    @property
    def lower_shadow(self): return min(self.o, self.c) - self.l
    @property
    def body(self): return abs(self.c - self.o)
    @property
    def direction(self):
        if self.c > self.o: return "阳"
        if self.c < self.o: return "阴"
        return "十字"


# =========================================================
# 3. 前日 OHLC → 关键位
# =========================================================

def compute_levels(bar: Bar) -> Dict[str, float]:
    po, ph, pl, pc = bar.o, bar.h, bar.l, bar.c
    r = ph - pl
    p = (ph + pl + pc) / 3
    return {
        "PH": ph, "PC": pc, "PO": po, "PL": pl,
        "entity_top": max(po, pc), "entity_bottom": min(po, pc),
        "entity_mid": (po + pc) / 2, "range_mid": (ph + pl) / 2,
        "fib_382": pl + 0.382*r, "fib_500": pl + 0.5*r,
        "fib_618": pl + 0.618*r, "fib_786": pl + 0.786*r,
        "pivot": p,
        "R1": 2*p - pl, "S1": 2*p - ph,
        "R2": p + r,      "S2": p - r,
        "limit_up": round(pc*1.10, 2),
        "limit_down": round(pc*0.90, 2),
    }


# =========================================================
# 4. 缠论：包含处理 / 分型 / 笔
# =========================================================

@dataclass
class MergedBar:
    dt_start: str; dt_end: str; h: float; l: float; direction: int

def merge_inclusion(bars: List[Bar]) -> List[MergedBar]:
    if not bars: return []
    merged = [MergedBar(bars[0].dt, bars[0].dt, bars[0].h, bars[0].l, 1)]
    direction = 1
    for bar in bars[1:]:
        last = merged[-1]
        contained = (bar.h <= last.h and bar.l >= last.l) or \
                    (bar.h >= last.h and bar.l <= last.l)
        if not contained:
            direction = 1 if (bar.h > last.h and bar.l > last.l) else -1
            merged.append(MergedBar(bar.dt, bar.dt, bar.h, bar.l, direction))
        else:
            if direction == 1:
                nh, nl = max(last.h, bar.h), max(last.l, bar.l)
            else:
                nh, nl = min(last.h, bar.h), min(last.l, bar.l)
            merged[-1] = MergedBar(last.dt_start, bar.dt, nh, nl, direction)
    return merged


class FractalType(Enum):
    TOP = "顶"; BOTTOM = "底"

@dataclass
class Fractal:
    type: FractalType; idx: int; h: float; l: float; dt: str

def find_fractals(merged: List[MergedBar]) -> List[Fractal]:
    out = []
    for i in range(1, len(merged)-1):
        p, c, n = merged[i-1], merged[i], merged[i+1]
        if c.h > p.h and c.h > n.h and c.l > p.l and c.l > n.l:
            out.append(Fractal(FractalType.TOP, i, c.h, c.l, c.dt_end))
        elif c.l < p.l and c.l < n.l and c.h < p.h and c.h < n.h:
            out.append(Fractal(FractalType.BOTTOM, i, c.h, c.l, c.dt_end))
    return out


@dataclass
class Stroke:
    start: Fractal; end: Fractal; direction: int

def build_strokes(fractals: List[Fractal], min_gap: int = 3) -> List[Stroke]:
    if not fractals: return []
    filt = [fractals[0]]
    for f in fractals[1:]:
        last = filt[-1]
        if f.type == last.type:
            if f.type == FractalType.TOP and f.h > last.h: filt[-1] = f
            elif f.type == FractalType.BOTTOM and f.l < last.l: filt[-1] = f
        else:
            filt.append(f)
    strokes = []
    i = 0
    while i < len(filt)-1:
        a, b = filt[i], filt[i+1]
        if abs(b.idx - a.idx) >= min_gap:
            strokes.append(Stroke(a, b, 1 if b.type == FractalType.TOP else -1))
            i += 1
        else:
            i += 1
    return strokes


@dataclass
class Center:
    zg: float; zd: float; start_idx: int; end_idx: int

def find_centers(strokes: List[Stroke], min_strokes: int = 3) -> List[Center]:
    centers = []
    i = 0
    while i + min_strokes - 1 < len(strokes):
        group = strokes[i:i+min_strokes]
        highs = [max(s.start.h, s.end.h) for s in group]
        lows  = [min(s.start.l, s.end.l) for s in group]
        zg, zd = min(highs), max(lows)
        if zg > zd:
            centers.append(Center(zg, zd, group[0].start.idx, group[-1].end.idx))
        i += 1
    return centers

# 5. 策略信号
# =========================================================

def generate_signals(df: pd.DataFrame,
                     stop_loss_pct: float = 0.05,
                     take_profit_pct: float = 0.10) -> pd.DataFrame:
    """
    策略逻辑：
      1. 用前日 OHLC 计算关键位
      2. 用缠论分型 + 笔 + 中枢判断结构
      3. 当日收盘价在关键位与中枢边界之间触发信号
    """
    signals = []
    position = 0  # 0 空仓，1 持仓
    entry_price = 0.0
    entry_date = None

    bars = [Bar(str(r.date.date()), r.open, r.high, r.low, r.close, r.volume)
            for r in df.itertuples()]

    for i in range(1, len(bars)):
        today = bars[i]
        prev  = bars[i-1]
        levels = compute_levels(prev)

        # 用近 30 根做缠论结构
        window = bars[max(0, i-30):i+1]
        merged = merge_inclusion(window)
        fracs = find_fractals(merged)
        strokes = build_strokes(fracs, min_gap=2)
        centers = find_centers(strokes, min_strokes=3)

        signal = "HOLD"

        if position == 0:
            # 买入条件示例：
            # 当前价站上前日实体上沿，且不在最后一个中枢内部
            above_entity = today.c > levels["entity_top"]
            above_center = True
            if centers:
                above_center = today.c > centers[-1].zg
            # 底分型确认
            has_bottom = any(f.type == FractalType.BOTTOM for f in fracs[-3:])
            if above_entity and above_center and has_bottom:
                signal = "BUY"
        else:
            # 卖出条件：止损 / 止盈 / 跌破中枢下沿
            pnl_pct = (today.c - entry_price) / entry_price
            broke_center = centers and today.c < centers[-1].zd
            if pnl_pct <= -stop_loss_pct:
                signal = "SELL_STOP"
            elif pnl_pct >= take_profit_pct:
                signal = "SELL_PROFIT"
            elif broke_center:
                signal = "SELL_STRUCT"

        # 执行
        if signal == "BUY" and position == 0:
            position = 1
            entry_price = today.c
            entry_date = today.dt
        elif signal.startswith("SELL") and position == 1:
            position = 0
            signals.append({
                "date": today.dt, "action": signal,
                "price": today.c, "entry": entry_price,
                "pnl_pct": (today.c - entry_price) / entry_price,
                "hold_days": i - bars.index(next(b for b in bars if b.dt == entry_date))
            })

    return pd.DataFrame(signals)

def backtest_stats(trades: pd.DataFrame, initial_capital: float = 100000.0):
    if trades.empty:
        return {"总交易次数": 0, "总收益": 0, "胜率": 0, "最大回撤": 0}

    equity = initial_capital
    peak = equity
    max_dd = 0.0

    for _, t in trades.iterrows():
        equity *= (1 + t["pnl_pct"])
        peak = max(peak, equity)
        dd = (peak - equity) / peak
        max_dd = max(max_dd, dd)

    total_return = (equity - initial_capital) / initial_capital
    win_rate = (trades["pnl_pct"] > 0).mean()
    avg_win = trades.loc[trades["pnl_pct"] > 0, "pnl_pct"].mean() if (trades["pnl_pct"] > 0).any() else 0
    avg_loss = trades.loc[trades["pnl_pct"] < 0, "pnl_pct"].mean() if (trades["pnl_pct"] < 0).any() else 0

    return {
        "总交易次数": len(trades),
        "总收益": f"{total_return:.2%}",
        "胜率": f"{win_rate:.2%}",
        "平均盈利": f"{avg_win:.2%}",
        "平均亏损": f"{avg_loss:.2%}",
        "最大回撤": f"{max_dd:.2%}",
        "最终资金": f"{equity:.2f}",
    }
